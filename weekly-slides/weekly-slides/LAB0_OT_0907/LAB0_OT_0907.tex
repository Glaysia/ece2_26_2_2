\begin{frame}{이번 학기에 배우는 내용}
\begin{itemize}
\item Verilog HDL로 회로를 기술하고 FPGA에 합성·구현한다.
\item C 프로그램을 컴파일하고 STM32에서 실행한다.
\item 입출력·타이머·PWM·통신 기능을 구현하거나 설정한다.
\item FPGA와 MCU 사이에 명령과 데이터를 주고받는다.
\item 두 장치를 활용한 팀 프로젝트를 구현하고 검증한다.
\end{itemize}
\end{frame}

\begin{frame}{왜 배워야 하는가?}
\begin{itemize}
\item 디지털논리의 조합회로·레지스터·FSM을 실제 보드에 구현한다.
\item C와 마이크로프로세서 지식을 주변장치 제어에 사용한다.
\item 핀에서 나오는 신호와 연결한 장치의 동작을 확인한다.
\item 요구사항을 정하고, 예상과 실제 결과를 비교하며 문제를 해결한다.
\end{itemize}
\end{frame}

\begin{frame}{전전컴 5개 분야와의 연결}
\begin{itemize}
\item 반도체·회로·디스플레이
\item 전기·제어·로봇
\item 컴퓨터·디지털시스템
\item 통신·신호처리
\item 전파·광·양자
\end{itemize}
\vspace{0.35cm}
디지털 회로 설계, 펌웨어, 통신과 계측의 기초를 연습합니다.
\end{frame}

\begin{frame}{반도체·회로·디스플레이}
\begin{itemize}
\item 디지털 회로 설계·검증: HDL 논리 설계와 파형 검증
\item 반도체 테스트·평가: 입출력 시험과 측정
\item 장비·디스플레이 제어 펌웨어: 입출력·타이머·통신
\end{itemize}
\vspace{0.35cm}
조합회로와 레지스터를 설계하고 시뮬레이션과 보드에서 확인합니다.
\end{frame}

\begin{frame}{전기·제어·로봇}
\begin{itemize}
\item 전력변환·모터 제어: ADC·PWM·인터럽트
\item 로봇 제어: 센서 입력, 통신과 동작 상태 제어
\item BMS 펌웨어: 측정, 조건 판단과 보호 상태 제어
\end{itemize}
\vspace{0.35cm}
입력을 읽고 조건에 맞는 제어 신호를 만드는 과정을 연습합니다.
\end{frame}

\begin{frame}{컴퓨터·디지털시스템}
\begin{itemize}
\item XPU·SoC·네트워크 칩 설계·검증: 레지스터·FSM·시뮬레이션
\item FPGA 개발: 논리 설계, 합성과 보드 검증
\item 펌웨어·드라이버 개발: 주변장치 설정과 인터럽트
\end{itemize}
\vspace{0.35cm}
회로의 동작과 소프트웨어의 제어를 함께 이해합니다.
\end{frame}

\begin{frame}{통신·신호처리}
\begin{itemize}
\item 모뎀·신호처리 회로: 데이터 경로와 타이밍 검증
\item 통신장치 펌웨어: 송수신, 데이터 형식과 상태 제어
\item 수업에서는 UART/SPI로 명령과 데이터를 전달한다.
\end{itemize}
\vspace{0.35cm}
두 보드가 같은 데이터 형식과 통신 규칙을 사용하도록 설계합니다.
\end{frame}

\begin{frame}{전파·광·양자}
\begin{itemize}
\item 레이더·RF·광학 장비 제어: 타이밍·통신·계측
\item 광송수신 모듈 펌웨어: 장치 설정과 상태 감시
\item 양자 제어 전자장치: 펄스·동작 순서·측정 인터페이스
\end{itemize}
\vspace{0.35cm}

\end{frame}

\end{itemize}
\end{frame}
